# 离线本地版智能文本处理系统实施计划

## 项目概述

将现有的基于 Docker + Ollama + Dify 的智能文档关键信息提取系统，改造为**完全离线、一键运行、低内存占用**的 Windows 桌面应用程序（.exe）。

### 当前架构问题
- **太重**：Docker + Ollama + Dify 至少占 6GB+ 内存
- **太复杂**：端口、容器、依赖、网络问题极多
- **无法打包 exe**：Docker 不能打包成单文件 exe
- **依赖网络**：需要连接 Dify API 服务

### 目标架构优势
| 原组件 | 轻量化替代 | 优势 |
|---|---|---|
| Ollama | llama.cpp / llama-cpp-python | 内存降低 70%，纯C++加速 |
| Dify 工作流 | Python 本地工作流引擎 | 无服务、无端口、可直接打包 |
| Docker | 无（直接裸跑） | 零部署成本 |
| 大模型 | 4-bit 量化小模型（Qwen2 1.8B/4B/7B） | 内存 512MB~1GB 就能跑 |
| Vue前端 | 保留UI + Python后端 | 复用现有界面设计 |

---

## 一、核心目标

1. **更智能**：支持本地知识库、语义理解、结构化输出、自动流程优化
2. **占用内存更少**：从 8GB+ 压到 **1GB 以内**
3. **部署更方便**：告别 Docker / 命令行 / 端口配置，双击即运行
4. **.exe 一键使用**：纯离线 Windows 可执行文件，无环境依赖、无需联网

```python
# engine/llm_engine.py
from llama_cpp import Llama
import os

class LocalLLMEngine:
    """本地轻量LLM引擎"""
    
    def __init__(self, model_path=None, n_ctx=2048, n_threads=None):
        """
        初始化本地LLM引擎
        
        Args:
            model_path: 模型文件路径(.gguf格式)
            n_ctx: 上下文窗口大小
            n_threads: 推理线程数，默认使用CPU核心数
        """
        self.model_path = model_path or self._get_default_model()
        self.n_ctx = n_ctx
        self.n_threads = n_threads or os.cpu_count()
        self.llm = None
        self._load_model()
    
    def _get_default_model(self):
        """获取默认模型路径"""
        base_dir = os.path.dirname(os.path.dirname(__file__))
        models_dir = os.path.join(base_dir, 'models')
        # 优先使用最小模型
        for model in ['qwen2-1.8b-4bit.gguf', 'qwen2-4b-4bit.gguf', 'llama3-2b-4bit.gguf']:
            path = os.path.join(models_dir, model)
            if os.path.exists(path):
                return path
        return None
    
    def _load_model(self):
        """加载模型"""
        if not self.model_path or not os.path.exists(self.model_path):
            raise FileNotFoundError(f"模型文件不存在: {self.model_path}")
        
        print(f"正在加载模型: {self.model_path}")
        self.llm = Llama(
            model_path=self.model_path,
            n_ctx=self.n_ctx,
            n_threads=self.n_threads,
            n_gpu_layers=0,  # 纯CPU运行，兼容所有电脑
            verbose=False
        )
        print("模型加载完成")
    
    def generate(self, prompt, max_tokens=512, temperature=0.1, stop=None):
        """
        生成文本
        
        Args:
            prompt: 输入提示词
            max_tokens: 最大生成token数
            temperature: 温度参数
            stop: 停止标记
        
        Returns:
            生成的文本
        """
        if not self.llm:
            raise RuntimeError("模型未加载")
        
        response = self.llm.create_completion(
            prompt=prompt,
            max_tokens=max_tokens,
            temperature=temperature,
            stop=stop or ["\n\n", "用户:", "Human:"],
        )
        return response["choices"][0]["text"].strip()
    
    def extract_info(self, text, instruction):
        """
        信息提取专用接口
        
        Args:
            text: 待处理文本
            instruction: 提取指令
        
        Returns:
            提取结果
        """
        prompt = f"""请根据以下指令从文档中提取信息：

指令：{instruction}

文档内容：
{text[:2000]}  # 限制输入长度，避免超出上下文

提取结果："""
        
        return self.generate(prompt, max_tokens=1024, temperature=0.1)
    
    def summarize(self, text, max_length=300):
        """文本总结"""
        prompt = f"请对以下文本进行简洁总结（不超过{max_length}字）：\n\n{text[:3000]}\n\n总结："
        return self.generate(prompt, max_tokens=max_length, temperature=0.1)
    
    def extract_keywords(self, text, count=10):
        """关键词提取"""
        prompt = f"请从以下文本中提取{count}个关键词，用逗号分隔：\n\n{text[:2000]}\n\n关键词："
        return self.generate(prompt, max_tokens=100, temperature=0.0)
```

```python
# engine/document_parser.py
import fitz  # PyMuPDF
from docx import Document
import pandas as pd
from PIL import Image
import pytesseract
import os

class DocumentParser:
    """多格式文档解析器"""
    
    SUPPORTED_FORMATS = {
        '.pdf': 'parse_pdf',
        '.docx': 'parse_docx',
        '.doc': 'parse_docx',
        '.txt': 'parse_txt',
        '.xlsx': 'parse_excel',
        '.xls': 'parse_excel',
        '.jpg': 'parse_image',
        '.jpeg': 'parse_image',
        '.png': 'parse_image',
    }
    
    @classmethod
    def parse(cls, file_path):
        """
        解析文档
        
        Args:
            file_path: 文档路径
        
        Returns:
            dict: 包含文本内容和元数据
        """
        ext = os.path.splitext(file_path)[1].lower()
        
        if ext not in cls.SUPPORTED_FORMATS:
            raise ValueError(f"不支持的文件格式: {ext}")
        
        parser_method = getattr(cls, cls.SUPPORTED_FORMATS[ext])
        return parser_method(file_path)
    
    @staticmethod
    def parse_pdf(file_path):
        """解析PDF文件"""
        doc = fitz.open(file_path)
        text = ""
        for page in doc:
            text += page.get_text()
        doc.close()
        return {
            'content': text,
            'type': 'pdf',
            'pages': len(doc),
            'filename': os.path.basename(file_path)
        }
    
    @staticmethod
    def parse_docx(file_path):
        """解析Word文件"""
        doc = Document(file_path)
        text = "\n".join([para.text for para in doc.paragraphs])
        return {
            'content': text,
            'type': 'docx',
            'paragraphs': len(doc.paragraphs),
            'filename': os.path.basename(file_path)
        }
    
    @staticmethod
    def parse_txt(file_path):
        """解析文本文件"""
        with open(file_path, 'r', encoding='utf-8') as f:
            text = f.read()
        return {
            'content': text,
            'type': 'txt',
            'filename': os.path.basename(file_path)
        }
    
    @staticmethod
    def parse_excel(file_path):
        """解析Excel文件"""
        df = pd.read_excel(file_path)
        text = df.to_string(index=False)
        return {
            'content': text,
            'type': 'excel',
            'rows': len(df),
            'columns': len(df.columns),
            'filename': os.path.basename(file_path)
        }
    
    @staticmethod
    def parse_image(file_path):
        """解析图片（OCR）"""
        image = Image.open(file_path)
        text = pytesseract.image_to_string(image, lang='chi_sim+eng')
        return {
            'content': text,
            'type': 'image',
            'filename': os.path.basename(file_path)
        }
```

```python
# engine/workflow.py
from typing import Dict, List, Callable, Any
from dataclasses import dataclass
from enum import Enum
import json

class NodeType(Enum):
    """节点类型"""
    INPUT = "input"           # 输入节点
    PARSE = "parse"           # 文档解析
    CLEAN = "clean"           # 文本清洗
    EXTRACT = "extract"       # 信息提取
    SUMMARIZE = "summarize"   # 文本总结
    CLASSIFY = "classify"     # 内容分类
    OUTPUT = "output"         # 输出节点
    CONDITION = "condition"   # 条件判断

@dataclass
class WorkflowNode:
    """工作流节点"""
    id: str
    type: NodeType
    name: str
    config: Dict[str, Any]
    next_nodes: List[str] = None
    
    def __post_init__(self):
        if self.next_nodes is None:
            self.next_nodes = []

class WorkflowEngine:
    """本地工作流引擎 - 替代Dify"""
    
    def __init__(self, llm_engine):
        """
        初始化工作流引擎
        
        Args:
            llm_engine: LLM引擎实例
        """
        self.llm = llm_engine
        self.nodes: Dict[str, WorkflowNode] = {}
        self.handlers: Dict[NodeType, Callable] = {
            NodeType.INPUT: self._handle_input,
            NodeType.PARSE: self._handle_parse,
            NodeType.CLEAN: self._handle_clean,
            NodeType.EXTRACT: self._handle_extract,
            NodeType.SUMMARIZE: self._handle_summarize,
            NodeType.CLASSIFY: self._handle_classify,
            NodeType.OUTPUT: self._handle_output,
            NodeType.CONDITION: self._handle_condition,
        }
    
    def add_node(self, node: WorkflowNode):
        """添加节点"""
        self.nodes[node.id] = node
    
    def execute(self, start_node_id: str, context: Dict[str, Any]) -> Dict[str, Any]:
        """
        执行工作流
        
        Args:
            start_node_id: 起始节点ID
            context: 执行上下文
        
        Returns:
            执行结果
        """
        current_node_id = start_node_id
        max_iterations = 100  # 防止无限循环
        iteration = 0
        
        while current_node_id and iteration < max_iterations:
            node = self.nodes.get(current_node_id)
            if not node:
                break
            
            handler = self.handlers.get(node.type)
            if handler:
                context = handler(node, context)
            
            # 移动到下一个节点
            if node.next_nodes:
                current_node_id = node.next_nodes[0]  # 简化处理，取第一个
            else:
                current_node_id = None
            
            iteration += 1
        
        return context
    
    def _handle_input(self, node: WorkflowNode, context: Dict) -> Dict:
        """处理输入节点"""
        return context
    
    def _handle_parse(self, node: WorkflowNode, context: Dict) -> Dict:
        """处理文档解析节点"""
        from .document_parser import DocumentParser
        
        file_path = context.get('file_path')
        if file_path:
            parsed = DocumentParser.parse(file_path)
            context['parsed_content'] = parsed
            context['text'] = parsed['content']
        return context
    
    def _handle_clean(self, node: WorkflowNode, context: Dict) -> Dict:
        """处理文本清洗节点"""
        text = context.get('text', '')
        # 基础清洗
        text = text.strip()
        text = ' '.join(text.split())  # 去除多余空白
        context['cleaned_text'] = text
        return context
    
    def _handle_extract(self, node: WorkflowNode, context: Dict) -> Dict:
        """处理信息提取节点"""
        text = context.get('text', '')
        instruction = node.config.get('instruction', '提取关键信息')
        
        result = self.llm.extract_info(text, instruction)
        context['extracted_info'] = result
        return context
    
    def _handle_summarize(self, node: WorkflowNode, context: Dict) -> Dict:
        """处理总结节点"""
        text = context.get('text', '')
        max_length = node.config.get('max_length', 300)
        
        result = self.llm.summarize(text, max_length)
        context['summary'] = result
        return context
    
    def _handle_classify(self, node: WorkflowNode, context: Dict) -> Dict:
        """处理分类节点"""
        text = context.get('text', '')
        categories = node.config.get('categories', [])
        
        prompt = f"请将以下文本分类到其中一个类别：{', '.join(categories)}\n\n文本：{text[:1000]}\n\n类别："
        result = self.llm.generate(prompt, max_tokens=50, temperature=0.0)
        context['category'] = result
        return context
    
    def _handle_output(self, node: WorkflowNode, context: Dict) -> Dict:
        """处理输出节点"""
        output_format = node.config.get('format', 'json')
        
        if output_format == 'json':
            context['output'] = json.dumps({
                'text': context.get('text', ''),
                'summary': context.get('summary', ''),
                'keywords': context.get('keywords', ''),
                'extracted_info': context.get('extracted_info', ''),
            }, ensure_ascii=False, indent=2)
        
        return context
    
    def _handle_condition(self, node: WorkflowNode, context: Dict) -> Dict:
        """处理条件节点"""
        condition = node.config.get('condition', '')
        # 简化实现
        return context
```

```python
            "Documents (*.pdf *.docx *.txt *.xlsx *.jpg *.png);;All Files (*)"
        )
        return json.dumps(files)
    
    @pyqtSlot(str, result=str)
    def process_document(self, file_path):
        """处理文档"""
        try:
            # 创建通用提取工作流
            start_node = WorkflowTemplates.create_document_extraction_workflow(
                self.workflow_engine
            )
            
            # 执行工作流
            context = {'file_path': file_path}
            result = self.workflow_engine.execute(start_node, context)
            
            return json.dumps({
                'success': True,
                'result': result
            }, ensure_ascii=False)
        except Exception as e:
            return json.dumps({
                'success': False,
                'error': str(e)
            }, ensure_ascii=False)
    
    @pyqtSlot(str, result=str)
    def get_model_info(self):
        """获取模型信息"""
        if self.llm_engine:
            return json.dumps({
                'loaded': True,
                'model_path': self.llm_engine.model_path,
                'n_ctx': self.llm_engine.n_ctx,
                'n_threads': self.llm_engine.n_threads
            })
        return json.dumps({'loaded': False})
```

```python
# tests/test_performance.py
import time
import psutil
import os

class PerformanceMonitor:
    """性能监控器"""
    
    def __init__(self):
        self.process = psutil.Process(os.getpid())
        self.start_time = None
        self.start_memory = None
    
    def start(self):
        """开始监控"""
        self.start_time = time.time()
        self.start_memory = self.process.memory_info().rss / 1024 / 1024  # MB
        print(f"初始内存占用: {self.start_memory:.1f} MB")
    
    def check(self, label=""):
        """检查当前状态"""
        elapsed = time.time() - self.start_time
        memory = self.process.memory_info().rss / 1024 / 1024  # MB
        print(f"[{label}] 运行时间: {elapsed:.1f}s, 内存占用: {memory:.1f} MB")
        return elapsed, memory
    
    def end(self):
        """结束监控"""
        elapsed, memory = self.check("完成")
        memory_increase = memory - self.start_memory
        print(f"内存增长: {memory_increase:.1f} MB")
        return elapsed, memory
```
